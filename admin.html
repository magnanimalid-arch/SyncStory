const CLAVE_SECRETA_ADMIN = "OroMusic2026"; const PIN_ADMIN = "5050"; function generarFirma(fechaStr) { let str = fechaStr + CLAVE_SECRETA_ADMIN; let hash = 0; for (let i = 0; i < str.length; i++) { const char = str.charCodeAt(i); hash = ((hash << 5) - hash) + char; hash |= 0; } return Math.abs(hash).toString(16).toUpperCase(); } function generar() { const pin = prompt("Ingrese el PIN de Administrador:"); if (pin !== PIN_ADMIN) { alert("PIN incorrecto."); return; } const hoy = new Date(); hoy.setDate(hoy.getDate() + 30); const yyyy = hoy.getFullYear(); const mm = String(hoy.getMonth() + 1).padStart(2, '0'); const dd = String(hoy.getDate()).padStart(2, '0'); const fechaExp = `${yyyy}${mm}${dd}`; const firma = generarFirma(fechaExp); const clave = `SYNC-${fechaExp}-${firma}`; const divRes = document.getElementById("resultado"); if (divRes) { divRes.style.display = "block"; divRes.innerHTML = "Clave Generada: " + clave; } prompt("Copia esta clave de 30 días:", clave); }
